# Floors, not exact pins.
#
# The original exact pins broke on Python 3.14: pydantic-core, psycopg-binary
# and PyYAML all shipped their first cp314 wheels well above the pinned
# versions, so pip fell back to building from source and failed. Each floor
# below is the first release with a Python 3.14 wheel, so the set resolves on
# 3.11 through 3.14.
#
# For byte-identical reproducibility in deployment, generate a lockfile from a
# working environment: pip freeze > requirements.lock

psycopg[binary]>=3.2.10,<4.0      # 3.2.10 = first cp314 wheel
psycopg-pool>=3.2.4,<4.0

PyYAML>=6.0.3,<7.0                # 6.0.3 = first cp314 wheel
python-dotenv>=1.0.1,<2.0
